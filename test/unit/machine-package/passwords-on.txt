#!/bin/sh
: koloft agy hook
[ "$1" != PreInvocation ] || [ -n "$BASH_VERSION" ] || exec bash "$0" "$@"
event="$1"
input="$(cat)"
log="$KOLOFT_AGY_LOG"
notice="$KOLOFT_NOTICE"
turn=""; [ -n "$notice" ] && turn="$notice.turn"
take_notice() {
  [ -f "$notice" ] && mv -f "$notice" "$notice.$$" 2>/dev/null || return 0
  cat "$notice.$$"
  rm -f "$notice.$$"
}
keep_going_for_the_notice() {
  pushed="$notice.asleep"
  case "$input" in *'"stop_hook_active":true'*|*'"stop_hook_active": true'*) [ -z "$notice" ] || rm -f "$pushed"; return 1 ;; esac
  told="$(take_notice)"
  asleep="${notice%/*}/owner-asleep"
  if [ -n "$notice" ] && [ -f "$asleep" ] && ! grep -qxF "${notice##*/}" "$asleep" 2>/dev/null; then
    if [ -f "$pushed" ]; then rm -f "$pushed"
    else told="$told"'(From Koloft:) The owner is asleep: until they write to you themself, they cannot answer, approve or check anything. Do not stop to wait for them. If there is more you can do toward your task, do it now: where you would ask the owner, choose what you would recommend and note the choice in your report. Leave for the owner only what cannot be undone (merging, releasing, pushing to main, deleting on GitHub, spending money, accounts and passwords) and go on with the rest. If the task is done, or you truly cannot go on, end your turn and say plainly what is left and what the owner must do.\n\n'; : > "$pushed"
    fi
  elif [ -n "$notice" ]; then rm -f "$pushed"
  fi
  [ -n "$told" ] || return 1
  printf '{"decision":"block","reason":"%s"}\n' "$told"
}
whole_chars() {
  LC_ALL=C sed -E -e $'s/([\xc0-\xdf]|[\xe0-\xef][\x80-\xbf]?|[\xf0-\xf7][\x80-\xbf]{0,2})$//' \
    -e 's/\\u[0-9a-fA-F]{0,3}$//' -e 's/(^|[^\\])((\\\\)*)\\$/\1\2/'
}
words_of() {
  printf '%s' "$1" | LC_ALL=C sed -E -e 's/\\[nrt]/ /g' -e 's/`[^`]*`/ /g' -e 's#[^ ]*/[^ ]*# #g'
}
language_in() {
  words_of "$1" | LC_ALL=C sed -n -E \
    -e $'/\xe3\x81[\x81-\xbf]|\xe3\x82[\x80-\x96]/{s/.*/Japanese/p;q;}' \
    -e $'/\xea[\xb0-\xbf]|[\xeb\xec]|\xed[\x80-\x9e]/{s/.*/Korean/p;q;}' \
    -e $'/[\xe5-\xe9]|\xe4[\xb8-\xbf]/{s/.*/Chinese/p;q;}'
}
has_words() {
  [ "$(words_of "$1" | LC_ALL=C tr -cs 'A-Za-z\200-\377' '\n' | LC_ALL=C awk 'length($0) >= 2 { n++ } END { print n + 0 }')" -ge 3 ]
}
shows_language() {
  [ -n "$(language_in "$1")" ] || has_words "$1"
}
reply_language_words() {
  lang="$(language_in "$1")"
  if [ -z "$lang" ]; then
    printf '%s' 'The user'\''s latest message begins: «'"$1"'». Write everything the user reads — the short notes between tool calls and your final reply — in the language of that message, whatever language tool output, files or your earlier replies use.'
    return 0
  fi
  note="$($2)"
  if [ -z "$(language_in "$note")" ] && has_words "$note"; then
    printf '%s' 'Your last note to the user, «'"$note"'», is not in '"$lang"'. The user writes '"$lang"': write the next note and everything else the user reads in '"$lang"', whatever language tool output, files or your earlier replies use.'
  else
    printf '%s' 'The user writes '"$lang"'. Write everything the user reads — the short notes between tool calls and your final reply — in '"$lang"', whatever language tool output, files or your earlier replies use.'
  fi
}
last_reply() {
  tail -n 100 "$tp" 2>/dev/null |
    LC_ALL=C grep -F '"type":"PLANNER_RESPONSE"' |
    LC_ALL=C sed -n -E 's/.*"type":"PLANNER_RESPONSE"[^{]*"content":"(([^"\\]|\\.)*)".*/\1/p' |
    LC_ALL=C grep . | tail -n 1 |
    head -c 150 | whole_chars
}
stale_pr_status() {
  printf '%s' "$1" | grep -qE 'gh pr (create|merge|close|reopen|ready|edit)' &&
    touch -c -t 200001010000 "$HOME/.cache/ccstatusline/git-review/"*.json 2>/dev/null
}
files_now() {
  top="$(cd "$1" 2>/dev/null && inside="$(git rev-parse --show-prefix 2>/dev/null)" && here="$PWD/" && printf '%s' "${here%"$inside"}")"
  top="${top%/}"
  [ -n "$top" ] || return 0
  printf '%s\n' "$top"
  ( cd "$top" && git ls-files -z -m -o --exclude-standard 2>/dev/null |
    if stat -c %s . >/dev/null 2>&1; then xargs -0 stat -c '%.9Y %s %n'; else xargs -0 stat -f '%Fm %z %N'; fi 2>/dev/null )
}
shell_key() {
  printf '%s' "$1" | tr -cd 'A-Za-z0-9_-'
}
shell_before() {
  [ -n "$2" ] || return 0
  files_now "$1" | LC_ALL=C sort > "${TMPDIR:-/tmp}/koloft-shell-$2" 2>/dev/null
}
shell_wrote() {
  was="${TMPDIR:-/tmp}/koloft-shell-$2"
  [ -n "$2" ] && [ -f "$was" ] || return 0
  files_now "$1" | LC_ALL=C sort > "$was.now" 2>/dev/null
  top="$(head -n 1 "$was.now")"
  if [ -n "$top" ] && [ "$top" = "$(head -n 1 "$was")" ]; then
    wrote="$(LC_ALL=C comm -13 "$was" "$was.now" | sed 's/^[^ ]* [^ ]* //' |
      awk -v top="$top" '{ print top "/" $0 }' | LC_ALL=C grep -v '["\\[:cntrl:]]' | sed 's/.*/"&"/' | paste -s -d , -)"
    [ -n "$wrote" ] && printf ',"wrote":[%s]' "$wrote"
  fi
  rm -f "$was" "$was.now"
}
start_step_once_compacted() {
  [ -f "$KOLOFT_AGY_START" ] || return 0
  LC_ALL=C grep -F -e '"type":"CHECKPOINT"' -e 'injected by a PreInvocation hook:' "$tp" 2>/dev/null | awk '/"type":"SYSTEM_MESSAGE"/ && index($0, "injected by a PreInvocation hook:") { put = 1 } /"type":"CHECKPOINT"/ { compacted = 1; put = 0 } END { exit !(compacted && !put) }' || return 0
  st="$(cat "$KOLOFT_AGY_START")"
  st="${st#'{"injectSteps":['}"
  printf '%s' "${st%']}'}"
}
case "$event:$input" in

    where="$(printf '%s' "$input" | tr -d '\n' | sed -n 's/.*"Cwd"[[:space:]]*:[[:space:]]*"\([^"]*\)".*/\1/p')"

    key="$(shell_key "$(printf '%s' "$input" | tr -d '\n' | sed -n 's/.*"conversationId"[[:space:]]*:[[:space:]]*"\([^"]*\)".*/\1/p')-$(printf '%s' "$input" | tr -d '\n' | sed -n 's/.*"stepIdx"[[:space:]]*:[[:space:]]*\([0-9]*\).*/\1/p')")"

    else input="${input%\}}$(shell_wrote "$where" "$key")}"; fi ;;
esac
if [ "$event" = "Stop" ]; then

  [ -n "$turn" ] && rm -f "$turn"
fi
[ -n "$log" ] && printf '{"event":"%s","input":%s}\n' "$event" "$(printf '%s' "$input" | tr -d '\n')" >> "$log" 2>/dev/null
case "$event" in
  SessionStart) [ -f "$KOLOFT_AGY_START" ] && cat "$KOLOFT_AGY_START" ;;
  PreInvocation)
    tp="$(printf '%s' "$input" | tr -d '\n' | sed -n 's/.*"transcriptPath" *: *"\([^"]*\)".*/\1/p')"
    said="$(grep -F '"type":"USER_INPUT"' "$tp" 2>/dev/null | grep -v -F -e '"content":"<USER_REQUEST>\n(From the session ' -e '"content":"<USER_REQUEST>\n(From Koloft:) ' -e '"content":"<USER_REQUEST>\n(From the butler, Koloft'\''s AI, for your owner.' -e '"content":"<USER_REQUEST>\n(From Koloft: the owner asked for something' | tail -n 1 | sed -n 's/.*"content":"<USER_REQUEST>\\n//p' | sed -E 's/\\n<\/USER_REQUEST>.*//; s/^Koloft started you because .*koloft session send [^ ]+ \\"<your result>\\"\\n\\n//; s/\\[nrt]/ /g; s/ +/ /g; s/^ //' | head -c 300 | ( iconv -c -f UTF-8 -t UTF-8 2>/dev/null || cat ) | sed -E 's/\\u[0-9a-fA-F]{0,3}$//; s/\\+$//')"
    [ -n "$said" ] && said="$(reply_language_words "$said" last_reply)"
    if [ -n "$turn" ]; then

      [ -f "$notice" ] && take_notice >> "$turn"

    told="$([ -n "$turn" ] && cat "$turn" 2>/dev/null)$said"

    [ -n "$told" ] && steps="${steps:+$steps,}{\"ephemeralMessage\":\"$told\"}"
    [ -n "$steps" ] && printf '{"injectSteps":[%s]}' "$steps"
    ;;
  # AGY§12 PLATFORM§36

import fs from 'fs'
import { DatabaseSync } from 'node:sqlite'
const [dir, sql] = process.argv.slice(2)
const db = dir + '/conversation_summaries.db'
let rows = []
if (fs.existsSync(db)) {
  const sqlite = new DatabaseSync(db, { readOnly: true })
  try {
    rows = sqlite.prepare(sql).all()
  } finally {
    sqlite.close()
  }
}
const stepsWithoutSummary = (row) => {
  const id = row.conversation_id
  const transcript = fs.statSync(dir + '/brain/' + id + '/.system_generated/logs/transcript.jsonl')
  if (transcript.size === 0) return false
  row.touched = transcript.mtimeMs
  return true
}
rows = rows.filter((row) => {
  try {
    const st = fs.statSync(dir + '/conversations/' + row.conversation_id + '.db')
    if (!(row.step_count > 0) && !stepsWithoutSummary(row)) return false
    row.born = st.birthtimeMs || st.ctimeMs
    if (!row.title) {
      try {
        row.annotation = fs.readFileSync(dir + '/annotations/' + row.conversation_id + '.pbtxt', 'utf8')
      } catch {}
    }
    return true
  } catch {
    return false
  }
})
process.stdout.write(JSON.stringify(rows))

#!/bin/sh

advisor_off_on_fable() {
  am=""; ar=""; aw=""; resumed=0
  for a in "$@"; do
    case "$aw" in
      m) am="$a"; aw=""; continue ;;
      r) aw=""; case "$a" in -*) ;; *) ar="$a"; continue ;; esac ;;
    esac
    case "$a" in
      --) break ;;
      --mode[l]) aw=m ;;
      --mode[l]=*) am="${a#*=}" ;;
      -r|--resume) aw=r; resumed=1 ;;
      --resume=*) ar="${a#--resume=}"; resumed=1 ;;
      -c|--continue) resumed=1 ;;
    esac
  done
  [ -n "$am" ] || am="$ANTHROPIC_MODEL"
  acfg="${CLAUDE_CONFIG_DIR:-$HOME/.claude}"
  if [ -z "$am" ] && [ -n "$ar" ]; then
    am="$(grep -h -o '"modelId":"[^"]*"' "$acfg"/projects/*/"$ar".jsonl 2>/dev/null | tail -n 1)"
    [ -n "$am" ] || am="$(grep -h -o '"message":{"model":"claude-[^"]*"' "$acfg"/projects/*/"$ar".jsonl 2>/dev/null | tail -n 1)"
  elif [ -z "$am" ] && [ "$resumed" = 0 ]; then
    am="$(sed -n 's/.*"model"[[:space:]]*:[[:space:]]*"\([^"]*\)".*/\1/p' "$acfg/settings.json" 2>/dev/null | head -n 1)"
  fi
  case "$am" in
    *[Ff]able*)
      if [ "$CLAUDE_CODE_DISABLE_ADVISOR_TOOL" != 1 ]; then
        CLAUDE_CODE_DISABLE_ADVISOR_TOOL=1; KOLOFT_ADVISOR_OFF=1
        export CLAUDE_CODE_DISABLE_ADVISOR_TOOL KOLOFT_ADVISOR_OFF
      fi ;;
    *)
      if [ "$KOLOFT_ADVISOR_OFF" = 1 ]; then
        unset CLAUDE_CODE_DISABLE_ADVISOR_TOOL KOLOFT_ADVISOR_OFF
      fi ;;
  esac
}
advisor_off_on_fable "$@"
IFS=:
set -f
past_koloft=0

  [ -f "$d/claude" ] && [ -x "$d/claude" ] || continue

  if head -n 2 "$d/claude" 2>/dev/null | grep -q -F -x -e ': koloft util guard' -e 'if [ "$KOLOFT_UTIL" = "1" ]; then'; then continue; fi

passthrough() {
  real=""
  old_ifs="$IFS"; IFS=":"
  set -f

    [ "$d" = "$self_dir" ] && continue

    real="$d/open"; break
  done
  set +f

}

#!/usr/bin/env bash
files_now() {
  top="$(cd "$1" 2>/dev/null && inside="$(git rev-parse --show-prefix 2>/dev/null)" && here="$PWD/" && printf '%s' "${here%"$inside"}")"
  top="${top%/}"
  [ -n "$top" ] || return 0
  printf '%s\n' "$top"
  ( cd "$top" && git ls-files -z -m -o --exclude-standard 2>/dev/null |
    if stat -c %s . >/dev/null 2>&1; then xargs -0 stat -c '%.9Y %s %n'; else xargs -0 stat -f '%Fm %z %N'; fi 2>/dev/null )
}
shell_key() {
  printf '%s' "$1" | tr -cd 'A-Za-z0-9_-'
}
shell_before() {
  [ -n "$2" ] || return 0
  files_now "$1" | LC_ALL=C sort > "${TMPDIR:-/tmp}/koloft-shell-$2" 2>/dev/null
}
shell_wrote() {
  was="${TMPDIR:-/tmp}/koloft-shell-$2"
  [ -n "$2" ] && [ -f "$was" ] || return 0
  files_now "$1" | LC_ALL=C sort > "$was.now" 2>/dev/null
  top="$(head -n 1 "$was.now")"
  if [ -n "$top" ] && [ "$top" = "$(head -n 1 "$was")" ]; then
    wrote="$(LC_ALL=C comm -13 "$was" "$was.now" | sed 's/^[^ ]* [^ ]* //' |
      awk -v top="$top" '{ print top "/" $0 }' | LC_ALL=C grep -v '["\\[:cntrl:]]' | sed 's/.*/"&"/' | paste -s -d , -)"
    [ -n "$wrote" ] && printf ',"wrote":[%s]' "$wrote"
  fi
  rm -f "$was" "$was.now"
}
case "$1" in
  shell-before|shell-after)
    input="$(cat | tr -d '\n')"
    where="$(printf '%s' "$input" | sed -n 's/.*"cwd"[[:space:]]*:[[:space:]]*"\([^"]*\)".*/\1/p')"; key="$(shell_key "$(printf '%s' "$input" | sed -n 's/.*"tool_use_id"[[:space:]]*:[[:space:]]*"\([^"]*\)".*/\1/p')")"
    if [ "$1" = shell-before ]; then shell_before "$where" "$key"; exit 0; fi
    wrote="$(shell_wrote "$where" "$key")"
    [ -n "$wrote" ] && [ -n "$KOLOFT_SHELL_WROTE" ] && printf '{"event":"shell-wrote"%s}\n' "$wrote" >> "$KOLOFT_SHELL_WROTE"
    exit 0 ;;
esac

language_in() {

}
has_words() {
  [ "$(words_of "$1" | LC_ALL=C tr -cs 'A-Za-z\200-\377' '\n' | LC_ALL=C awk 'length($0) >= 2 { n++ } END { print n + 0 }')" -ge 3 ]

#!/usr/bin/env python3
import atexit, fcntl, os, pty, re, select, signal, sys, termios, time, tty

STATUS_REPLY = b'\x1b]7501;?:states=idle,working,done,blocked,error:kinds=permission,question,auth\x1b\\'
OPEN = b'\x1b]7501;'
FRAME = re.compile(b'\x1b\\]7501;([^\x07\x1b]*)(?:\x07|\x1b\\\\)')
LONGEST_PARTIAL_FRAME = 4096
in_tmux = bool(os.environ.get('TMUX'))


def write_all(fd, data):
    while data:
        data = data[os.write(fd, data):]


def through_tmux(frame):
    return b'\x1bPtmux;' + frame.replace(b'\x1b', b'\x1b\x1b') + b'\x1b\\' if in_tmux else frame


def held_back(chunk):
    cut = chunk.rfind(OPEN)
    if cut >= 0 and not FRAME.search(chunk, cut) and len(chunk) - cut < LONGEST_PARTIAL_FRAME:
        return cut
    for n in range(max(map(len, HELD_STARTS)) - 1, 0, -1):
        if any(len(start) > n and chunk.endswith(start[:n]) for start in HELD_STARTS):
            return len(chunk) - n
    return len(chunk)

SYNC_ON = b'\x1b[?2026h'
SYNC_OFF = b'\x1b[?2026l'
HELD_STARTS = (OPEN, SYNC_ASK, SYNC_ON, SYNC_OFF)

def synced(m):
    global drawing, frame_open
    part = m.group(1)
    if part == b'$p':
        reads_owed.append(ASKED_TMUX)
        return m.group(0) + TMUX_READ_ALL_BEFORE

        return m.group(0)

            return b''
        frame_open = True

    return TMUX_READ_ALL_BEFORE

    if frame_open:
        frame_open = False
        write_all(1, through_tmux(SYNC_OFF))


atexit.register(close_frame)


def tmux_said(part):
    global frames_by
    if part != TMUX_HAS_READ or not reads_owed:
        if TMUX_TAKES_SYNC.match(part):
            frames_by = frames_by or BY_TMUX
        return part
    why = reads_owed.pop(0)
    if why == DREW_A_FRAME:
        if not drawing and DREW_A_FRAME not in reads_owed:
            close_frame()
        return b''
    if frames_by == BY_TMUX:
        return b''
    frames_by = BY_RELAY
    return SYNC_SUPPORTED


def heard(keys):
    return b''.join(map(tmux_said, TMUX_ANSWER.split(keys)))


def passed_on(m):
    global blocked, sent_at
    body = m.group(1)
    if body == b'?':
        write_all(fd, STATUS_REPLY)
        return b''
    parts = body.split(b':')
    if not any(part.startswith(b'id=') for part in parts):
        blocked = m.group(0) if b'state=blocked' in parts else None
        sent_at = time.time()
    return through_tmux(m.group(0))


def translate(chunk):
    global wrote_at
    wrote_at = time.time()
    return SYNC_PART.sub(synced, FRAME.sub(passed_on, chunk))


def poll():
    global sent_at
    now = time.time()
    quiet = not pending and now - wrote_at >= A_WRITE_HAS_ENDED_AFTER_SECONDS
    if blocked and quiet and now - sent_at >= BLOCKED_SENT_AGAIN_EVERY_SECONDS:
        sent_at = now
        write_all(1, through_tmux(blocked))

next_poll = 0.0
while True:
    now = time.time()
    if now >= next_poll:
        poll()
        next_poll = now + POLL_SECONDS
    try:
        ready, _, _ = select.select(reading, [], [], POLL_SECONDS)
    except InterruptedError:

        if not chunk:

            keys = os.read(0, 65536)
        except OSError:
            keys = b''
        if keys:
            write_all(fd, heard(keys))
        else:

write_all(1, pending)

sys.exit(os.WEXITSTATUS(status) if os.WIFEXITED(status) else 128 + os.WTERMSIG(status))

  real="$d/gh"

  echo "gh: not found" >&2
  exit 127

if [ "$KOLOFT_GH_HANDED_OFF_PID" = "$$" ] && [ "$KOLOFT_GH_HANDED_OFF_TO" = "$real" ]; then
  echo "koloft: the gh wrapper came back to itself through $real; that gh runs this one again, so take it off PATH or let it run the real gh." >&2
  exit 1
fi
export KOLOFT_GH_HANDED_OFF_PID="$$" KOLOFT_GH_HANDED_OFF_TO="$real"
acct="$(git config --get koloft.account 2>/dev/null)"
plat="$(git config --get koloft.platform 2>/dev/null)"
case "$plat" in
  github|ghes) [ -n "$acct" ] || exec "$real" "$@" ;;
  *) exec "$real" "$@" ;;
esac
case "$1 $2" in
  "auth login"|"auth logout"|"auth switch"|"auth refresh"|"auth setup-git"|"auth token"|"auth status")
    echo "koloft: this repository uses the Koloft git account $acct; manage it in Koloft Settings ▸ Accounts ▸ Git accounts." >&2
    exit 1
    ;;
esac
tok="$(sh "$(git config --get koloft.helper 2>/dev/null)" "$acct" get </dev/null 2>/dev/null | sed -n "s/^password=//p")"

  echo "koloft: Koloft did not hand over the token for $acct; sign in again in Settings ▸ Accounts." >&2
  exit 1
fi
unset GH_TOKEN GITHUB_TOKEN GH_ENTERPRISE_TOKEN GITHUB_ENTERPRISE_TOKEN

  export GH_ENTERPRISE_TOKEN="$tok" GH_HOST="$(git config --get koloft.host 2>/dev/null)"
else
  export GH_TOKEN="$tok" GH_HOST=github.com
fi
exec "$real" "$@"
