#!/bin/sh
: koloft git credential helper
proto=""
host=""
while IFS= read -r line; do
  case "$line" in
    protocol=*) proto="${line#protocol=}" ;;
    host=*) host="${line#host=}" ;;
  esac
done
case "$proto$host" in *[\"\\]*) proto="" host="" ;; esac
dir='/fixture/git-requests'
newid() {
  u="$(uuidgen 2>/dev/null | tr 'A-Z' 'a-z')"
  if [ -z "$u" ] && [ -r /proc/sys/kernel/random/uuid ]; then u="$(cat /proc/sys/kernel/random/uuid)"; fi
  echo "$u"
}
unreachable() {
  echo "koloft: could not reach Koloft for git credentials." >&2
  exit 1
}
send() {
  id="$(newid)"
  [ -n "$id" ] || id="$$-$(date +%s)"
  part="$dir/git-$id.part"
  if ! { printf '{"kind":"%s","account":"%s","protocol":"%s","host":"%s"}\n' "$1" "$2" "$proto" "$host" > "$part"; } 2>/dev/null || ! mv "$part" "$dir/git-$id.json" 2>/dev/null; then
    rm -f "$part" 2>/dev/null
    unreachable
  fi
}
ask() {
  send "$1" "$2"
  res="$dir/gitres-$id.json"
  deadline=$(( $(date +%s) + 10 ))
  while [ ! -f "$res" ] && [ "$(date +%s)" -lt "$deadline" ]; do sleep 0.05; done
  if [ ! -f "$res" ]; then
    rm -f "$dir/git-$id.json" 2>/dev/null
    unreachable
  fi
  reply="$(cat "$res" 2>/dev/null)"
  rm -f "$res" 2>/dev/null
}
case "$2" in
  get)
    ask credential "$1"
    user="$(printf '%s' "$reply" | LC_ALL=C sed -n 's/.*"username":"\([^"]*\)".*/\1/p')"
    pw="$(printf '%s' "$reply" | LC_ALL=C sed -n 's/.*"password":"\([^"]*\)".*/\1/p')"
    [ -n "$pw" ] || exit 0
    printf 'username=%s\npassword=%s\n' "$user" "$pw"
    ;;
  erase) send rejected "$1" ;;
esac
exit 0
