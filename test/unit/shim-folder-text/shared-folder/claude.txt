    printf '   %s\n' 'Non-interactive use is fine: claude -p · --help · doctor · mcp · …' >&2
    exit 1
  fi
fi

# CC§16
claude_meets_minimum() {
  kmin_v="$("$1" --version 2>/dev/null | sed -n '1s/^[^0-9]*\([0-9][0-9]*\.[0-9][0-9]*\.[0-9][0-9]*\).*/\1/p')"
  [ -z "$kmin_v" ] && return 0
  awk -v a="$kmin_v" -v b="2.1.293" 'BEGIN { split(a, x, "."); split(b, y, "."); for (i = 1; i <= 3; i++) if (x[i] + 0 != y[i] + 0) exit (x[i] + 0 < y[i] + 0); exit 0 }'
}
ensure_claude_minimum() {
  claude_meets_minimum "$1" && return 0
  printf '[Koloft] Claude Code %s is older than %s, the oldest this Koloft supports. Updating it now...\n' "$kmin_v" "2.1.293" >&2
  "$1" update
  claude_meets_minimum "$1" && return 0
  printf '[Koloft] Claude Code is still older than %s. Run "claude update" yourself, then start the session again.\n' "2.1.293" >&2
  return 1
}
if [ "$skip" = "0" ]; then ensure_claude_minimum "$real" || exit 1; fi

newid() {
  u="$(uuidgen 2>/dev/null | tr 'A-Z' 'a-z')"
  if [ -z "$u" ] && [ -r /proc/sys/kernel/random/uuid ]; then u="$(cat /proc/sys/kernel/random/uuid)"; fi
  echo "$u"
}

register() {
  [ "$KOLOFT_UTIL" = "1" ] && return 0
  reg_dir="$KOLOFT_SESSION_DIR"
  if [ -z "$reg_dir" ]; then reg_dir="$HOME/.koloft/sessions"; fi
  mkdir -p "$reg_dir"
  printf '{"tabId":"%s","regId":"%s","sessionId":"%s","cwd":"%s","ts":%s,"mode":"%s","pid":%s}\n' \
    "$KOLOFT_TAB_ID" "$1" "$2" "$PWD" "$(date +%s)" "$3" "$$" > "$reg_dir/$1.json"
}

refuse() { echo "koloft: $1" >&2; exit 78; }
pick_field() { printf '%s\n' "$pjson" | sed -n 's/.*"'"$1"'":"\([^"]*\)".*/\1/p'; }
pick_account() {
  [ -n "$KOLOFT_PICK_DIR" ] && [ -n "$KOLOFT_TAB_ID" ] && [ -n "$KOLOFT_PID" ] && kill -0 "$KOLOFT_PID" 2>/dev/null ||
    refuse "$1 runs here only inside a Koloft tab, on an account from Settings ▸ Accounts."
  pickid="$(newid)"
  [ -n "$pickid" ] || pickid="$$-$(date +%s)"
  preq="$KOLOFT_PICK_DIR/req-$pickid.json"
  pres="$KOLOFT_PICK_DIR/res-$pickid.json"
  printf '{"tabId":"%s","ts":%s,"backend":"%s"}\n' "$KOLOFT_TAB_ID" "$(date +%s)" "$1" > "$preq" 2>/dev/null
  pwaits=0
  while [ ! -f "$pres" ] && [ "$pwaits" -lt 60 ]; do sleep 0.05; pwaits=$((pwaits+1)); done
  pjson="$(cat "$pres" 2>/dev/null)"
  rm -f "$preq" "$pres" 2>/dev/null
  pacct="$(pick_field account)"
  if [ -z "$pacct" ]; then
    [ -z "$pjson" ] && refuse "picking an account took too long. Try again."
    refuse "$2"
  fi
}
# PLATFORM§3 PLATFORM§2 ADR-0001
read_keychain() {
  ptokf="$KOLOFT_PICK_DIR/tok-$pickid"
  ( umask 077; exec security find-generic-password -s "$1" -a "$2" -w > "$ptokf" 2>/dev/null ) &
  psec=$!
  pswait=0
  while kill -0 "$psec" 2>/dev/null && [ "$pswait" -lt 100 ]; do sleep 0.05; pswait=$((pswait+1)); done
  kill "$psec" 2>/dev/null
  wait "$psec" 2>/dev/null
  ptok="$(cat "$ptokf" 2>/dev/null)"
  rm -f "$ptokf" 2>/dev/null
}

advisor_off_on_fable() {
  am=""; ar=""; aw=""; resumed=0
  for a in "$@"; do
    case "$aw" in
      m) am="$a"; aw=""; continue ;;
      r) aw=""; case "$a" in -*) ;; *) ar="$a"; continue ;; esac ;;
    esac
    case "$a" in
      --) break ;;
      --mode[l]) aw=m ;;
      --mode[l]=*) am="${a#*=}" ;;
      -r|--resume) aw=r; resumed=1 ;;
      --resume=*) ar="${a#--resume=}"; resumed=1 ;;
      -c|--continue) resumed=1 ;;
    esac
  done
  [ -n "$am" ] || am="$ANTHROPIC_MODEL"
  acfg="${CLAUDE_CONFIG_DIR:-$HOME/.claude}"
  if [ -z "$am" ] && [ -n "$ar" ]; then
    am="$(grep -h -o '"modelId":"[^"]*"' "$acfg"/projects/*/"$ar".jsonl 2>/dev/null | tail -n 1)"
    [ -n "$am" ] || am="$(grep -h -o '"message":{"model":"claude-[^"]*"' "$acfg"/projects/*/"$ar".jsonl 2>/dev/null | tail -n 1)"
  elif [ -z "$am" ] && [ "$resumed" = 0 ]; then
    am="$(sed -n 's/.*"model"[[:space:]]*:[[:space:]]*"\([^"]*\)".*/\1/p' "$acfg/settings.json" 2>/dev/null | head -n 1)"
  fi
  case "$am" in
    *[Ff]able*)
      if [ "$CLAUDE_CODE_DISABLE_ADVISOR_TOOL" != 1 ]; then
        CLAUDE_CODE_DISABLE_ADVISOR_TOOL=1; KOLOFT_ADVISOR_OFF=1
        export CLAUDE_CODE_DISABLE_ADVISOR_TOOL KOLOFT_ADVISOR_OFF
      fi ;;
    *)
      if [ "$KOLOFT_ADVISOR_OFF" = 1 ]; then
        unset CLAUDE_CODE_DISABLE_ADVISOR_TOOL KOLOFT_ADVISOR_OFF
      fi ;;
  esac
}
