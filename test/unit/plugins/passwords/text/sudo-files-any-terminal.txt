=== sudo
#!/bin/sh
: koloft sudo wrapper
self="$(cd "$(dirname "$0")" >/dev/null 2>&1 && pwd)"
real=""
IFS=:
set -f
for d in $PATH; do
  [ "$d" = "$self" ] && continue
  [ -f "$d/sudo" ] && [ -x "$d/sudo" ] || continue
  head -n 2 "$d/sudo" 2>/dev/null | grep -q "koloft sudo wrapper" && continue
  real="$d/sudo"
  break
done
set +f
unset IFS
if [ -z "$real" ]; then
  echo "sudo: not found" >&2
  exit 127
fi
if [ -n "$SUDO_ASKPASS" ] || (exec </dev/tty) 2>/dev/null; then exec "$real" "$@"; fi
for a in "$@"; do
  case "$a" in
    -S|--stdin|-n|--non-interactive) exec "$real" "$@" ;;
    --) break ;;
    -*) ;;
    *) break ;;
  esac
done
SUDO_ASKPASS="$self/koloft-askpass" exec "$real" -A "$@"
=== koloft-askpass
#!/bin/sh
: koloft askpass
exec "$(dirname "$0")/koloft" secret sudo "$1" "$(ps -o args= -p "$PPID" 2>/dev/null)"
