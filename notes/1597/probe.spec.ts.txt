const OUT = process.env.PROBE_OUT!
const note = (line: string): void => fs.appendFileSync(OUT, `${line}\n`)

test.describe.configure({ timeout: 600_000, retries: 0 })
test.beforeAll(() => test.skip(!dockerAvailable(), 'needs docker'))

const asRoot = (lab: SshLab, script: string): string =>
  execFileSync('docker', ['exec', lab.target, 'sh', '-c', script], { encoding: 'utf8' })

function toolOnlyOnRcPath(lab: SshLab, user: string, rc: string, line: string): void {
  asRoot(
    lab,
    "mkdir -p /opt/ztools/bin && printf '#!/bin/sh\\necho ztool-ran\\n' > /opt/ztools/bin/ztool && chmod 755 /opt/ztools/bin/ztool"
  )
  runOnTarget(lab, user, `mkdir -p "$(dirname ${rc})" && printf '\\n%s\\n' '${line}' >> ${rc}`)
}

const EXPORT_LINE = 'export PATH="/opt/ztools/bin:$PATH"'

async function withLab(
  env: E2EEnv,
  before: (lab: SshLab) => void,
  body: (ctx: { app: ElectronApplication; page: Page; lab: SshLab }) => Promise<void>
): Promise<void> {
  installCodex(env, JSON.stringify({ tokens: { access_token: 't', account_id: 'lab' } }))
  const lab = await startSshLab(env)
  let app: ElectronApplication | null = null
  try {
    installLabSsh(env, lab)
    installFakeCodexOnTarget(lab)
    before(lab)
    app = await launchApp(env)
    const page = await app.firstWindow()
    await page.waitForLoadState('domcontentloaded')
    await waitBooted(page)
    await body({ app, page, lab })
  } finally {
    if (app) await quitAndClose(app)
    stopSshLab(lab, defaultControlDir())
  }
}

async function probe(
  page: Page,
  lab: SshLab,
  label: string,
  alias: LabAlias = 'kt-key',
  user = 'kuser',
  claudeToo = true
): Promise<void> {
  expect((await addWorkspace(page, remoteKeyFor(alias, user))).code).toBe('added')
  await expect(wsGroup(page, alias).locator('.ws-conn')).toHaveClass(/\bon\b/, { timeout: 45_000 })
  const started = Date.now()
  await newSessionInWith(page, alias, 'Codex')
  await expect(wsRows(page, alias).filter({ hasText: 'Codex fixture session' })).toHaveClass(
    /st-waiting/,
    { timeout: 180_000 }
  )
  note(`${label} codex tab bound after ${Date.now() - started} ms`)
  const tabId = (await page.evaluate(() => window.api.sessions.list())).find(
    (s) => s.backendId === 'codex' && s.alive
  )!.tabId
  await page.evaluate(
    ([id, typed]) => window.api.terminal.write(id, typed),
    [tabId, '/sh command -v ztool\r']
  )
  let codexExit: string | undefined
  await expect
    .poll(
      async () =>
        (codexExit = /sh exit=(\d+)[\s\S]*Codex done/.exec(await terminalText(page, tabId))?.[1]),
      { timeout: 60_000 }
    )
    .toBeDefined()
  note(`${label} codex: command -v ztool exit=${codexExit}`)
  note(
    `${label} codex server ${runOnTarget(lab, user, 'tr "\\0" "\\n" < /proc/$(pgrep -u $(id -u) -f "app-server --listen" | head -n1)/environ | grep "^PATH="').trim()}`
  )
  note(
    `${label} leftover tmp files: ${runOnTarget(lab, user, 'ls /tmp | grep -c "^tmp[.]" || true').trim()}`
  )
  if (!claudeToo) return
  await startSessionIn(page, alias, { remote: true, method: 'Claude' })
  await runIn(page, centerTerm(page), '/sh command -v ztool')
  await expect(centerTerm(page)).toContainText(/sh exit=\d+/, { timeout: 60_000 })
  note(
    `${label} claude: command -v ztool ${/sh exit=\d+/.exec((await centerTerm(page).textContent()) ?? '')?.[0]}`
  )
}

test('probe: bash user, PATH line at the end of ~/.bashrc', async ({ env }) => {
  await withLab(
    env,
    (lab) => toolOnlyOnRcPath(lab, 'kuser', '.bashrc', EXPORT_LINE),
    ({ page, lab }) => probe(page, lab, 'bash/.bashrc-end')
  )
})

test('probe: zsh user, PATH line in ~/.zshrc, which also echoes and leaves a job behind', async ({
  env
}) => {
  await withLab(
    env,
    (lab) => {
      asRoot(
        lab,
        'apt-get update -qq && DEBIAN_FRONTEND=noninteractive apt-get install -y -qq --no-install-recommends zsh >/dev/null && usermod -s /usr/bin/zsh kuser'
      )
      toolOnlyOnRcPath(lab, 'kuser', '.zshrc', EXPORT_LINE)
      runOnTarget(lab, 'kuser', "printf 'echo hello-from-zshrc\\nsleep 25 &\\n' >> .zshrc")
    },
    ({ page, lab }) => probe(page, lab, 'zsh/.zshrc+echo+bg-job')
  )
})
